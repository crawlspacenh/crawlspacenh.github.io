{{/* Responsive WebP image from assets/. Input dict: src ("/images/posters/x.jpg"), alt, width (display px, default 400), class, href ("" = no link, "original" = link to full-size original, else a URL). Only JPG/PNG/GIF/WebP are processed; anything else (PDF, SVG, ...) is skipped with a warning. */}} {{ $img := resources.Get (strings.TrimPrefix "/" (.src | default "")) }} {{ if not $img }} {{ warnf "picture.html: image not found in assets: %q" .src }} {{ else if not (in (slice "jpeg" "png" "gif" "webp") $img.MediaType.SubType) }} {{ warnf "picture.html: %q is not a JPG/PNG/GIF/WebP image; skipping" .src }} {{ else }} {{ $orig := $img }} {{ if ne $img.MediaType.SubType "gif" }} {{ $img = $img.Filter images.AutoOrient }} {{ end }} {{ $w := int (.width | default 400) }} {{ $w1 := cond (gt $img.Width $w) $w $img.Width }} {{ $one := $img.Resize (printf "%dx webp q80" $w1) }} {{ $srcset := printf "%s 1x" $one.RelPermalink }} {{ if gt $img.Width (mul $w 2) }} {{ $two := $img.Resize (printf "%dx webp q80" (mul $w 2)) }} {{ $srcset = printf "%s, %s 2x" $srcset $two.RelPermalink }} {{ end }} {{ $alt := .alt | default "" }} {{ $class := .class | default "" }} {{ if eq .href "original" }} {{ else if .href }} {{ else }} {{ end }} {{ end }}
